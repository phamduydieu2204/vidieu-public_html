📊 CẤU TRÚC DATABASE
1. Quản lý Accounts & Pools
vd_provider_accounts - Tài khoản gốc
sqlCREATE TABLE vd_provider_accounts (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  
  -- Thông tin cơ bản
  provider VARCHAR(64) NOT NULL,              -- 'helium10', 'freepik'...
  account_login VARCHAR(191) NOT NULL,        -- Email đăng nhập gốc
  display_name VARCHAR(191) NOT NULL,         -- 'Helium10 Slot #1'
  status ENUM('active','suspended','expired') DEFAULT 'active',
  capacity INT UNSIGNED NOT NULL DEFAULT 1,   -- Bao nhiêu license chia sẻ
  
  -- THÔNG TIN TÀI KHOẢN (plaintext)
  cookie TEXT NULL,
  cookie_format ENUM('json','netscape','headers') DEFAULT 'json',
  cookie_updated_at DATETIME NULL,
  
  login_email VARCHAR(255) NULL,
  login_password VARCHAR(255) NULL,
  totp_secret VARCHAR(100) NULL,              -- Mã 2FA
  recovery_email VARCHAR(255) NULL,
  recovery_phone VARCHAR(50) NULL,
  
  notes TEXT NULL,
  meta JSON NULL,
  
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  UNIQUE KEY uq_provider_login (provider, account_login)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
Ví dụ dữ liệu:

vd_product_pools - Nhóm accounts cho sản phẩm
sqlCREATE TABLE vd_product_pools (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  product_id BIGINT UNSIGNED NOT NULL,        -- WooCommerce product ID
  name VARCHAR(191) NOT NULL,                 -- 'Pool Helium10 Platinum'
  strategy ENUM('sticky','weighted','priority') DEFAULT 'sticky',
  
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  UNIQUE KEY uq_product_pool (product_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
Chiến lược phân bổ:

sticky: License gắn cố định 1 account (ổn định)
weighted: Phân bổ theo capacity (account lớn nhận nhiều hơn)
priority: Theo thứ tự ưu tiên (A đầy → B → C...)


vd_pool_accounts - Liên kết accounts vào pool
sqlCREATE TABLE vd_pool_accounts (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  pool_id BIGINT UNSIGNED NOT NULL,           -- → vd_product_pools
  provider_account_id BIGINT UNSIGNED NOT NULL, -- → vd_provider_accounts
  
  capacity_override INT NULL,                 -- Ghi đè capacity nếu cần
  weight INT NULL,                            -- Dùng cho weighted strategy
  priority TINYINT NULL,                      -- Dùng cho priority strategy
  
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  INDEX (pool_id),
  INDEX (provider_account_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

vd_cookie_assignments - Gán license → account
sqlCREATE TABLE vd_cookie_assignments (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  license_id BIGINT UNSIGNED NOT NULL,        -- → lmfwc_licenses.id
  provider_account_id BIGINT UNSIGNED NOT NULL, -- → vd_provider_accounts
  
  strategy ENUM('sticky','round_robin','priority') DEFAULT 'sticky',
  assigned_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  status ENUM('active','paused','migrating','revoked') DEFAULT 'active',
  notes TEXT NULL,
  
  UNIQUE KEY uq_license_assignment (license_id, status),
  INDEX (provider_account_id, status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
Ý nghĩa:

Bản đồ: License #123 đang dùng Account #5
Admin có thể đổi account cho license (migrate)
Chỉ có 1 assignment active mỗi lúc

sqlCREATE TABLE vd_product_share_configs (

3. Kiểm soát thiết bị
vd_device_fingerprints - Danh bạ thiết bị
sqlCREATE TABLE vd_device_fingerprints (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  fp_hash CHAR(64) NOT NULL UNIQUE,           -- SHA-256 hash
  
  first_seen DATETIME DEFAULT CURRENT_TIMESTAMP,
  last_seen DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  ua VARCHAR(512) NULL,                       -- User-Agent
  os VARCHAR(64) NULL,                        -- Windows 10, Android...
  browser VARCHAR(64) NULL,                   -- Chrome, Safari...
  device_model VARCHAR(64) NULL,              -- iPhone 13...
  
  meta JSON NULL                              -- screen, lang, tz...
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

vd_license_devices - Allowlist thiết bị cho license
sqlCREATE TABLE vd_license_devices (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  license_id BIGINT UNSIGNED NOT NULL,
  device_fp CHAR(64) NOT NULL,                -- → fp_hash
  
  status ENUM('pending','approved','blocked') DEFAULT 'pending',
  approved_by BIGINT UNSIGNED NULL,           -- Admin user ID
  approved_at DATETIME NULL,
  last_used_at DATETIME NULL,
  
  notes TEXT NULL,
  
  UNIQUE KEY uq_license_device (license_id, device_fp),
  INDEX (license_id, status),
  INDEX (device_fp)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

vd_license_device_limits - Hạn mức thiết bị
sqlCREATE TABLE vd_license_device_limits (
  license_id BIGINT UNSIGNED PRIMARY KEY,     -- 1 license = 1 config
  
  max_devices INT UNSIGNED NOT NULL DEFAULT 1, -- Tối đa bao nhiêu thiết bị
  rotation_cooldown_seconds INT UNSIGNED DEFAULT 86400, -- 24h cooldown
  
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
Ví dụ:
sql-- License VIP: 3 thiết bị, cooldown 12h
INSERT INTO vd_license_device_limits VALUES (123, 3, 43200, NOW());

-- License thường: 1 thiết bị, cooldown 24h
INSERT INTO vd_license_device_limits VALUES (456, 1, 86400, NOW());

4. Logging & Audit Trail
vd_account_fetch_log - Lịch sử lấy thông tin
sqlCREATE TABLE vd_account_fetch_log (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  license_id BIGINT UNSIGNED NOT NULL,
  device_fp CHAR(64) NOT NULL,
  provider_account_id BIGINT UNSIGNED NOT NULL,
  
  fetched_fields JSON NOT NULL,               -- ["cookie","login_email"]
  
  fetched_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  ip_address VARCHAR(45) NULL,
  user_agent TEXT NULL,
  status ENUM('success','fail','blocked','rate_limited') DEFAULT 'success',
  reason TEXT NULL,
  
  INDEX (license_id, fetched_at),
  INDEX (device_fp, fetched_at),
  INDEX (provider_account_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

vd_license_access_log - Lịch sử verify device
sqlCREATE TABLE vd_license_access_log (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  license_id BIGINT UNSIGNED NOT NULL,
  device_fp CHAR(64) NOT NULL,
  
  action ENUM('verify','approve','block','rotate') NOT NULL,
  status ENUM('success','fail','pending','blocked') NOT NULL,
  reason TEXT NULL,
  
  ip_address VARCHAR(45) NULL,
  ip_country CHAR(2) NULL,
  ip_city VARCHAR(100) NULL,
  
  accessed_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  INDEX (license_id, accessed_at),
  INDEX (device_fp, accessed_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

vd_license_rate_limits - Chống spam request
sqlCREATE TABLE vd_license_rate_limits (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  license_id BIGINT UNSIGNED NOT NULL,
  
  window_seconds INT UNSIGNED NOT NULL DEFAULT 300,  -- 5 phút
  max_hits INT UNSIGNED NOT NULL DEFAULT 10,         -- 10 lần
  last_reset DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  UNIQUE KEY uq_license_window (license_id, window_seconds)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

🔌 REST API
1. POST /wp-json/vd/v1/device/verify
Mục đích: Xác thực license + thiết bị
Request:
httpPOST /wp-json/vd/v1/device/verify
Content-Type: application/json

{
  "license_key": "XXXX-XXXX-XXXX",
  "fp": "sha256_fingerprint",
  "ua": "Mozilla/5.0...",
  "os": "Windows 10",
  "screen": "1920x1080"
}
Response:
json{
  "ok": true,
  "deviceStatus": "approved",
  "remainingSlots": 2,
  "cooldownEndsAt": null,
  "license": {
    "expires_at": "2025-11-06T00:00:00Z",
    "days_remaining": 30
  }
}
Trường hợp lỗi:
json{
  "ok": false,
  "deviceStatus": "blocked",
  "reason": "Device not in allowlist"
}

4. POST /wp-json/vd/v1/device/rotate
Mục đích: Yêu cầu đổi thiết bị
Request:
httpPOST /wp-json/vd/v1/device/rotate
Content-Type: application/json

{
  "license_key": "XXXX-XXXX-XXXX",
  "old_device_fp": "sha256_old",
  "new_device_fp": "sha256_new",
  "reason": "Mất máy cũ"
}
Response (thành công):
json{
  "ok": true,
  "message": "Device rotation successful",
  "cooldown_until": "2025-10-07T10:30:00Z"
}
Response (bị cooldown):
json{
  "ok": false,
  "error": "Cooldown not passed",
  "cooldown_ends_at": "2025-10-07T10:30:00Z",
  "hours_remaining": 18
}

Hàm lấy thông tin account cho license
php<?php
/**
 * Lấy thông tin account cho license
 * Luôn trả về thông tin MỚI NHẤT từ database
 */
function vd_get_account_info_for_license($license_key, $device_fp) {
    global $wpdb;
    
    // 1. Verify license + device
    $license = vd_verify_license($license_key, $device_fp);
    if (!$license['ok']) {
        return $license; // Trả về lỗi
    }
    
    // 2. Check rate limit
    if (!vd_check_rate_limit($license['license_id'])) {
        return [
            'ok' => false,
            'error' => 'Rate limit exceeded',
            'retry_after' => 300
        ];
    }
    
    // 3. Lấy assignment (license -> provider_account)
    $assignment = $wpdb->get_row($wpdb->prepare("
        SELECT provider_account_id 
        FROM vd_cookie_assignments 
        WHERE license_id = %d AND status = 'active'
        LIMIT 1
    ", $license['license_id']));
    
    if (!$assignment) {
        return [
            'ok' => false,
            'error' => 'No account assigned to this license'
        ];
    }
    
    // 4. Lấy thông tin account MỚI NHẤT
    $account = $wpdb->get_row($wpdb->prepare("
        SELECT * 
        FROM vd_provider_accounts 
        WHERE id = %d AND status = 'active'
    ", $assignment->provider_account_id), ARRAY_A);
    
    if (!$account) {
        return [
            'ok' => false,
            'error' => 'Account not found or inactive'
        ];
    }
    
    // 5. Lấy cấu hình chia sẻ của sản phẩm
    $config = $wpdb->get_row($wpdb->prepare("
        SELECT share_fields, instructions, display_format
        FROM vd_product_share_configs 
        WHERE product_id = %d
    ", $license['product_id']));
    
    if (!$config) {
        // Nếu chưa cấu hình, mặc định chỉ chia sẻ cookie
        $share_fields = ['cookie'];
        $instructions = 'Sử dụng cookie được cung cấp';
    } else {
        $share_fields = json_decode($config->share_fields, true);
        $instructions = $config->instructions;
    }
    
    // 6. Filter chỉ lấy fields được phép
    $account_info = [];
    
    if (in_array('cookie', $share_fields) && !empty($account['cookie'])) {
        $account_info['cookie'] = [
            'value' => $account['cookie'],
            'format' => $account['cookie_format'],
            'updated_at' => $account['cookie_updated_at']
        ];
    }
    
    if (in_array('login_email', $share_fields) && !empty($account['login_email'])) {
        $account_info['login_email'] = $account['login_email'];
    }
    
    if (in_array('login_password', $share_fields) && !empty($account['login_password'])) {
        $account_info['login_password'] = $account['login_password'];
    }
    
    if (in_array('totp_secret', $share_fields) && !empty($account['totp_secret'])) {
        $account_info['totp_secret'] = $account['totp_secret'];
    }
    
    if (in_array('recovery_email', $share_fields) && !empty($account['recovery_email'])) {
        $account_info['recovery_email'] = $account['recovery_email'];
    }
    
    if (in_array('recovery_phone', $share_fields) && !empty($account['recovery_phone'])) {
        $account_info['recovery_phone'] = $account['recovery_phone'];
    }
    
    // 7. Log lại lần fetch
    vd_log_account_fetch(
        $license['license_id'],
        $device_fp,
        $assignment->provider_account_id,
        $share_fields
    );
    
    // 8. Trả về kết quả
    return [
        'ok' => true,
        'license' => [
            'key' => $license_key,
            'product_name' => get_the_title($license['product_id']),
            'expires_at' => $license['expires_at'],
            'days_remaining' => $license['days_remaining']
        ],
        'account_info' => $account_info,
        'instructions' => $instructions
    ];
}

/**
 * Verify license + device
 */
function vd_verify_license($license_key, $device_fp) {
    global $wpdb;
    
    // Lấy license từ LMfWC
    $license = lmfwc_get_license($license_key);
    
    if (!$license) {
        return ['ok' => false, 'error' => 'Invalid license key'];
    }
    
    // Check expire
    $expires_at = strtotime($license->expires_at);
    $now = time();
    
    if ($expires_at < $now) {
        return ['ok' => false, 'error' => 'License expired'];
    }
    
    // Check device
    $device = $wpdb->get_row($wpdb->prepare("
        SELECT status 
        FROM vd_license_devices 
        WHERE license_id = %d AND device_fp = %s
    ", $license->id, $device_fp));
    
    if (!$device) {
        // Device chưa có trong DB → tạo mới với status pending
        $wpdb->insert('vd_license_devices', [
            'license_id' => $license->id,
            'device_fp' => $device_fp,
            'status' => 'pending'
        ]);
        
        return [
            'ok' => false,
            'error' => 'Device not approved',
            'device_status' => 'pending'
        ];
    }
    
    if ($device->status !== 'approved') {
        return [
            'ok' => false,
            'error' => 'Device not approved',
            'device_status' => $device->status
        ];
    }
    
    // Update last_used_at
    $wpdb->update('vd_license_devices', 
        ['last_used_at' => current_time('mysql')],
        ['license_id' => $license->id, 'device_fp' => $device_fp]
    );
    
    // Success
    return [
        'ok' => true,
        'license_id' => $license->id,
        'product_id' => $license->product_id,
        'expires_at' => $license->expires_at,
        'days_remaining' => ceil(($expires_at - $now) / 86400)
    ];
}

/**
 * Check rate limit
 */
function vd_check_rate_limit($license_id) {
    global $wpdb;
    
    $limit = $wpdb->get_row($wpdb->prepare("
        SELECT window_seconds, max_hits, last_reset 
        FROM vd_license_rate_limits 
        WHERE license_id = %d
    ", $license_id));
    
    if (!$limit) {
        // Chưa có config → cho phép
        return true;
    }
    
    $now = time();
    $last_reset = strtotime($limit->last_reset);
    $elapsed = $now - $last_reset;
    
    if ($elapsed > $limit->window_seconds) {
        // Reset window
        $wpdb->update('vd_license_rate_limits',
            ['last_reset' => current_time('mysql')],
            ['license_id' => $license_id]
        );
        return true;
    }
    
    // Check số lần gọi trong window
    $hits = $wpdb->get_var($wpdb->prepare("
        SELECT COUNT(*) 
        FROM vd_account_fetch_log 
        WHERE license_id = %d 
          AND fetched_at > %s
    ", $license_id, date('Y-m-d H:i:s', $last_reset)));
    
    return $hits < $limit->max_hits;
}

/**
 * Log fetch
 */
function vd_log_account_fetch($license_id, $device_fp, $provider_account_id, $fields) {
    global $wpdb;
    
    $wpdb->insert('vd_account_fetch_log', [
        'license_id' => $license_id,
        'device_fp' => $device_fp,
        'provider_account_id' => $provider_account_id,
        'fetched_fields' => json_encode($fields),
        'fetched_at' => current_time('mysql'),
        'ip_address' => $_SERVER['REMOTE_ADDR'],
        'user_agent' => $_SERVER['HTTP_USER_AGENT'],
        'status' => 'success'
    ]);
}
